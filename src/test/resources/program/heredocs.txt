#!/usr/bin/env bash

name="${USER_NAME:-developer}"
version="${APP_VERSION:-1.0.0}"

cat <<EOF
application:
  name: clash
  version: $version
  owner: $name
EOF

cat <<'SQL' > query.sql
SELECT id, name
FROM users
WHERE active = true
ORDER BY name;
SQL

read -r -d '' message <<-TEXT
	Hello $name,
	The generated version is $version.
TEXT
